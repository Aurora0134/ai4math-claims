    #!/usr/bin/env python3
    import re, sys
    sys.stdout.reconfigure(encoding="utf-8")
    SNAP = r"<REPO-ROOT>\tasks\<source-task-id>\formalized\nfact-primorial-square.statement.txt"
    FINAL = r"<REPO-ROOT>\tasks\<source-task-id>\attempts\final-nfact-primorial-square.lean"
    NAMES = ["isSquare_factorial_mul_primorial_iff","exists_sq_factorial_mul_primorial_iff",
             "isSquare_iff_all_factorization_odd","exists_prime_between_div_three_and_div_two",
             "witnesses_zero_to_five","primorial_factorization_eq_one",
             "primorial_factorization_eq_zero","factorization_factorial_mul_primorial"]
    strip = lambda s: re.sub(r"--[^\n]*","",re.sub(r"/-.*?-/","",s,flags=re.S))
    collapse = lambda s: re.sub(r"\s+"," ",strip(s)).strip()
    squeeze = lambda s: re.sub(r"\s+","",strip(s))
    hits = lambda t,n: [m.start() for m in re.finditer(r"(?m)^theorem "+re.escape(n)+r"\b",t)]
    block = re.search(r"```lean\n(.*?)```",open(SNAP,encoding="utf-8").read(),flags=re.S).group(1)
    fin = open(FINAL,encoding="utf-8").read()
    ok = True
    for n in NAMES:
        assert len(hits(block,n))==1 and len(hits(fin,n))==1, n   # uniqueness
        i = hits(block,n)[0]; j2 = block.find("\n/--",i)
        seg_s = block[i:] if j2==-1 else block[i:j2]
        j = hits(fin,n)[0]; seg_f = fin[j:fin.find(":=",j)]
        r = (collapse(seg_s)==collapse(seg_f), squeeze(seg_s)==squeeze(seg_f))
        ok &= all(r); print(("MATCH" if all(r) else "DIFF"), n)
    print("VERDICT:", "PASS" if ok else "FAIL")
